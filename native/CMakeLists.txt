cmake_minimum_required(VERSION 3.24)
project(mirrorme_native LANGUAGES C CXX)
if(NOT WIN32 OR NOT MINGW OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "MirrorMe native requires an x64 MinGW-w64 UCRT toolchain.")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/../lib")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/../tests")
set(PROTOCOL_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../build/native/protocol" CACHE PATH "Pinned patched protocol archive")
set(NATIVE_SDK_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../build/native/sdk/ucrt64" CACHE PATH "Static crypto/plist SDK")
set(AUDIO_SDK_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../build/native-audio/install" CACHE PATH "Static FFmpeg SDK")
option(MM_PROTOCOL_ONLY "Build protocol with an explicit test-only media stub; not a production receiver" OFF)
add_compile_definitions(WIN32 WINVER=0x0A00 _WIN32_WINNT=0x0A00 NTDDI_VERSION=0x0A000000
  OPENSSL_API_COMPAT=0x10101000L LIBPLIST_STATIC)
add_compile_options(-ffunction-sections -fdata-sections)
add_link_options(-static -static-libgcc -static-libstdc++ -Wl,--gc-sections -Wl,--no-insert-timestamp)

foreach(archive crypto plist-2.0)
  if(NOT EXISTS "${NATIVE_SDK_DIR}/lib/lib${archive}.a")
    message(FATAL_ERROR "Missing static archive lib${archive}.a. Run scripts/build-native.ps1.")
  endif()
endforeach()
add_subdirectory("${PROTOCOL_SOURCE_DIR}/lib/playfair" playfair)
add_subdirectory("${PROTOCOL_SOURCE_DIR}/lib/llhttp" llhttp)
file(GLOB protocol_sources CONFIGURE_DEPENDS "${PROTOCOL_SOURCE_DIR}/lib/*.c")
list(FILTER protocol_sources EXCLUDE REGEX "/dnssd\\.c$")
add_library(airplay STATIC ${protocol_sources})
target_include_directories(airplay PRIVATE "${PROTOCOL_SOURCE_DIR}/lib/playfair"
  "${PROTOCOL_SOURCE_DIR}/lib/llhttp" "${NATIVE_SDK_DIR}/include")
target_compile_definitions(airplay PRIVATE STANDALONE TARGET_POSIX _REENTRANT
  _LARGEFILE64_SOURCE _FILE_OFFSET_BITS=64 NOHOLD PLIST_210 PLIST_230)
target_link_libraries(airplay PUBLIC playfair llhttp
  "${NATIVE_SDK_DIR}/lib/libplist-2.0.a" "${NATIVE_SDK_DIR}/lib/libcrypto.a"
  winpthread ws2_32 iphlpapi crypt32 bcrypt advapi32)
add_library(mirrorme STATIC receiver.cpp dnssd_windows.cpp)
target_compile_definitions(mirrorme PRIVATE UNICODE _UNICODE)
target_include_directories(mirrorme PRIVATE "${PROTOCOL_SOURCE_DIR}/lib" "${NATIVE_SDK_DIR}/include")
target_include_directories(mirrorme PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(mirrorme PUBLIC airplay dnsapi)

if(NOT MM_PROTOCOL_ONLY)
  foreach(source media.cpp audio_decode.cpp)
    if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${source}")
      message(FATAL_ERROR "Production source ${source} is not available; MM_PROTOCOL_ONLY is test-only.")
    endif()
  endforeach()
  target_sources(mirrorme PRIVATE media.cpp audio_decode.cpp)
  target_include_directories(mirrorme PRIVATE "${AUDIO_SDK_DIR}/include")
  foreach(archive avcodec avutil)
    if(NOT EXISTS "${AUDIO_SDK_DIR}/lib/lib${archive}.a")
      message(FATAL_ERROR "Run scripts/build-audio-codecs.ps1 to build static lib${archive}.a.")
    endif()
    target_link_libraries(mirrorme PUBLIC "${AUDIO_SDK_DIR}/lib/lib${archive}.a")
  endforeach()
  target_link_libraries(mirrorme PUBLIC mfplat mf mfuuid wmcodecdspuuid
    d3d11 dxgi dxguid ole32 oleaut32 uuid user32 gdi32 avrt propsys shlwapi ksuser bcrypt)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(protocol_receive_bounds tests/protocol_receive_bounds.cpp)
  target_include_directories(protocol_receive_bounds PRIVATE "${PROTOCOL_SOURCE_DIR}/lib")
  target_link_libraries(protocol_receive_bounds PRIVATE airplay)
  target_compile_options(protocol_receive_bounds PRIVATE -Wall -Wextra -Werror)
  target_link_options(protocol_receive_bounds PRIVATE "-Wl,--wrap=malloc,--wrap=calloc,--wrap=realloc")
  add_test(NAME protocol_receive_bounds COMMAND protocol_receive_bounds)
  set_tests_properties(protocol_receive_bounds PROPERTIES TIMEOUT 30)
  add_executable(protocol_tests tests/protocol_tests.cpp)
  if(MM_PROTOCOL_ONLY)
    target_sources(protocol_tests PRIVATE tests/protocol_media_stub.cpp)
    target_compile_definitions(protocol_tests PRIVATE MM_PROTOCOL_TEST_STUB)
  else()
    enable_language(RC)
    target_sources(protocol_tests PRIVATE tests/protocol_resource.rc)
    target_include_directories(protocol_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/tests")
    add_executable(protocol_media_invariants tests/protocol_media_invariants.cpp tests/protocol_resource.rc)
    target_include_directories(protocol_media_invariants PRIVATE
      "${PROTOCOL_SOURCE_DIR}/lib" "${NATIVE_SDK_DIR}/include" "${CMAKE_CURRENT_SOURCE_DIR}/tests")
    target_compile_definitions(protocol_media_invariants PRIVATE UNICODE _UNICODE
      PROTOCOL_VIDEO_FIXTURE="${CMAKE_CURRENT_SOURCE_DIR}/../receiver/tests/blue-frame.h264")
    target_link_libraries(protocol_media_invariants PRIVATE mirrorme)
    add_test(NAME protocol_media_invariants COMMAND protocol_media_invariants)
    set_tests_properties(protocol_media_invariants PROPERTIES TIMEOUT 30)
    add_executable(media_flicker tests/media_flicker_probe.cpp
      tests/media_audio_stub.cpp tests/protocol_resource.rc)
    target_include_directories(media_flicker PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/tests")
    target_compile_definitions(media_flicker PRIVATE MM_MEDIA_TESTING UNICODE _UNICODE)
    target_compile_options(media_flicker PRIVATE -Wall -Wextra -Werror)
    target_link_libraries(media_flicker PRIVATE mfplat mfuuid ole32 oleaut32 uuid user32 gdi32 ksuser)
    add_test(NAME media_flicker COMMAND media_flicker)
    set_tests_properties(media_flicker PROPERTIES TIMEOUT 30
      WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/..")
  endif()
  target_include_directories(protocol_tests PRIVATE "${PROTOCOL_SOURCE_DIR}/lib" "${NATIVE_SDK_DIR}/include")
  target_link_libraries(protocol_tests PRIVATE mirrorme)
  add_test(NAME protocol_lifecycle COMMAND protocol_tests)
  set_tests_properties(protocol_lifecycle PROPERTIES TIMEOUT 90)
endif()
